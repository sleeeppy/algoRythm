```Python
from operator import concat

n = int(input())
cycle = 0

def Cycle(num:int) -> int:
    rightSum = 0;
    
    if(num < 10):
        rightSum = str(num)
        # concat('0', str(num)[-1:])
    else:
        # rightSum = str(int(str(num)[0]) + int(str(num)[1]))
        rightSum = str(sum(map(int, str(num))))

    global cycle;
    cycle +=1

    return int(concat(str(num)[-1:], rightSum[-1:]))

cur = Cycle(n)

while(n != cur):
    cur = Cycle(cur)
else:
    print(cycle)
```
