- 참고로 원소는 < 1000, n<=100

```java
import java.io.*;
import java.util.*;

public class Main {
  public static void main(String[] args) throws Exception {
        String[] s = IO.readln().split(" ");
        int n = Integer.parseInt(s[0]);
    
        int[] arr = new int[n+1];
        for(int i = 1; i <= n; i++)
          arr[i] = Integer.parseInt(s[i]);
    
        ArrayList<ArrayList<Integer>> list = new ArrayList<>();
        list.add(new ArrayList<>());
        int idx = 0;

        for(int i = 1; i <= n; i++)
        {
            if(arr[i - 1] > arr[i])
            {
                list.add(new ArrayList<>());
                idx++;
            }
            list.get(idx).add(arr[i]);
        }

        int count = 0;
        int sum = 0;

        for(int i = 0; i < list.size(); i++)
        {
            if(list.get(i).size() == 1) continue;

            count++;

            for(int j = 0; j < list.get(i).size(); j++)
            {
                sum += list.get(i).get(j);   
            }
        }

        IO.println(count + " " + sum);
  }
}
```

```java
import java.io.*;
import java.util.*;

public class Main {
  public static void main(String[] args) throws Exception {
    String[] s = IO.readln().split(" ");
    int n = Integer.parseInt(s[0]);
    int[] arr = new int[n+2];

    for(int i = 1; i <= n; i++) 
      arr[i] = Integer.parseInt(s[i]);

    int sum = 0;
    int count = 0;

    for(int i = 1; i <= n;)
    {
      int j = i;
      while(j < n && arr[j] <= arr[j+1]) j++;

      if(i == j)
      {
        i++;
        continue;
      }

      count++;

      for(int k = i; k <= j; k++)
        sum += arr[k];

      i = j + 1;

    }

    IO.println(count + " " + sum);

  }
}
```
